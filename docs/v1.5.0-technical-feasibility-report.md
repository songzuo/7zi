### 1.2 现有机制评估

// 1. 添加 jitter 到重连延迟
function getReconnectDelay(attempt: number, baseDelay: number = 1000): number {
  const exponentialDelay = Math.min(baseDelay * Math.pow(1.5, attempt), 30000);
  const jitter = Math.random() * 0.3 * exponentialDelay; // 0-30% 抖动
  return Math.floor(exponentialDelay + jitter);
}

  // 客户端错误
  VALIDATION = 'VALIDATION_ERROR',
  NETWORK = 'NETWORK_ERROR',
  TIMEOUT = 'TIMEOUT',

  // HTTP 错误
  BAD_REQUEST = 'BAD_REQUEST',
  UNAUTHORIZED = 'UNAUTHORIZED',
  FORBIDDEN = 'FORBIDDEN',
  NOT_FOUND = 'NOT_FOUND',
  RATE_LIMIT = 'RATE_LIMIT_EXCEEDED',
  INTERNAL = 'INTERNAL_ERROR',
  SERVICE_UNAVAILABLE = 'SERVICE_UNAVAILABLE',
}

// 可重试错误判断
export function isRetryable(error: AppError): boolean {
  const retryableTypes = [
    AppErrorType.NETWORK,
    AppErrorType.TIMEOUT,
    AppErrorType.SERVICE_UNAVAILABLE,
    AppErrorType.RATE_LIMIT,
  ]

  return retryableTypes.includes(error.type) || error.retryable === true
}

// 错误严重等级
export enum ErrorSeverity {
  LOW = 'low', // 静默处理
  MEDIUM = 'medium', // toast 通知
  HIGH = 'high', // 显著提示 + 可选重试
  CRITICAL = 'critical', // 全屏错误 + 强制操作
}

// 基础 Store 创建器
interface StoreConfig<S> {
  name: string
  initialState: S
  persist?: PersistConfig
  middleware?: Middleware[]
}

function createTypedStore<S>(config: StoreConfig<S>) {
  return create<StoreWithMiddleware<S>>()(
    devtools(
      persist((set, get) => ({ ...config.initialState }), {
        name: config.name,
        ...config.persist,
      }),
      { name: config.name }
    )
  )
}

// 使用示例
const dashboardStore = createTypedStore({
  name: 'dashboard',
  initialState: {
    members: [],
    issues: [],
    isLoading: false,
  },
  persist: {
    partialize: state => ({
      members: state.members,
    }),
  },
})
