```typescript
// RBAC权限系统
interface Permission {
  resource: string;    // 'tasks', 'agents', 'settings'
  action: string;      // 'read', 'write', 'delete', 'admin'
  conditions?: Record<string, unknown>;
}

interface Role {
  id: string;
  name: string;
  permissions: Permission[];
}

interface User {
  id: string;
  roles: string[];
  permissions: Permission[]; // 计算得出
}

// 权限检查中间件
function checkPermission(
  user: User,
  resource: string,
  action: string
): boolean {
  return user.permissions.some(p => 
    p.resource === resource && 
    (p.action === action || p.action === '*')
  );
}
```

```typescript
class WorkflowEngine {
  async execute(workflow: Workflow, input: unknown): Promise<unknown> {
    let currentNode = workflow.nodes.find(n => n.type === 'input');
    let result = input;
    
    while (currentNode && currentNode.type !== 'output') {
      // 执行节点逻辑
      result = await this.executeNode(currentNode, result);
      
      // 查找下一个节点
      const edge = workflow.edges.find(e => e.source === currentNode!.id);
      if (!edge) break;
      
      currentNode = workflow.nodes.find(n => n.id === edge.target);
    }
    
    return result;
  }
  
  private async executeNode(node: WorkflowNode, input: unknown): Promise<unknown> {
    switch (node.type) {
      case 'agent':
        return await this.executeAgentTask(node, input);
      case 'condition':
        return await this.evaluateCondition(node, input);
      default:
        return input;
    }
  }
}
```
